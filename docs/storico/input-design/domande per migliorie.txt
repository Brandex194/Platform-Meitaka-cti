function _generateNarrative(customer, matches, mlScore) {

  if (!matches.length) {

      <strong>Assessment Summary</strong><br><br>
      Nessuna vulnerabilità critica nota rilevata
      nell'inventario corrente del cliente.
      Si raccomanda monitoraggio continuo,
      patch management regolare e revisione esposizione.
    `;
  }

  const critical = matches.filter(m => m.sev === 'critical').length;
  const internet = matches.filter(m => m.internetExposed).length;

  return `
    <strong>Assessment Summary</strong><br><br>
